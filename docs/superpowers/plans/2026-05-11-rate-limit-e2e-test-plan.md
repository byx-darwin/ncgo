> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `ncgo test rate-limit` (seed + run) commands and support `path = '*'` wildcard in rate-limit SQL queries, so users can test rate-limiting end-to-end via Docker Compose.

**Architecture:** `ncgo test rate-limit seed` generates SQL and pipes it through `docker compose exec postgres psql`. `ncgo test rate-limit run` invokes vegeta (local or via docker compose). The compose.yaml template (generated by `internal/scaffold/shared/container.go`) already supports postgres/redis; we add vegeta. SQL wildcard support lives in `internal/assets/_data/hertz/layout.yaml`.

**Tech Stack:** Go (Cobra CLI), Docker Compose, PostgreSQL, Redis, vegeta, sqlc.

```go
// Package ratelimit provides rate-limit test seeding via docker compose exec.
package ratelimit

import (
	"context"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"

	"github.com/byx-darwin/ncgo/internal/manifest"
)

// SeedOptions configures the seed operation.
type SeedOptions struct {
	Root        string
	MaxRequests int
	WindowSecs  int
}

// Seed inserts test rate-limit rules into PostgreSQL via docker compose exec.
func Seed(ctx context.Context, opts SeedOptions) error {
	if opts.Root == "" {
		opts.Root = "."
	}
	m, err := manifest.Load(opts.Root)
	if err != nil {
		return fmt.Errorf("load manifest: %w", err)
	}
	if opts.MaxRequests == 0 {
		opts.MaxRequests = 10
	}
	if opts.WindowSecs == 0 {
		opts.WindowSecs = 60
	}

	service := m.Service.Name
	if service == "" {
		service = filepath.Base(opts.Root)
	}

	sql := buildSeedSQL(service, opts.MaxRequests, opts.WindowSecs)
	return execSQLViaDockerCompose(ctx, opts.Root, sql)
}

func buildSeedSQL(service string, maxRequests, windowSecs int) string {
	return fmt.Sprintf(`DELETE FROM rate_limit_rules WHERE service = '%s';

INSERT INTO rate_limit_rules (service, phase, method, match_kind, path,
    enabled, key_by, strategy, window_seconds, max_requests)
VALUES
    ('%s', 'pre_auth', '*', 'exact', '*', true, ARRAY['ip'], 'fixed_window', %d, %d),
    ('%s', 'pre_auth', 'GET', 'exact', '/healthz', true, ARRAY['ip'], 'fixed_window', %d, %d);
`, service, service, windowSecs, maxRequests, service, windowSecs, maxRequests)
}

func execSQLViaDockerCompose(ctx context.Context, root, sql string) error {
	if _, err := exec.LookPath("docker"); err != nil {
		return fmt.Errorf("docker not found on PATH. Install docker and run 'docker compose up -d' first")
	}

	cmd := exec.CommandContext(ctx, "docker", "compose", "exec", "-T", "postgres", "psql", "-U", "postgres", "-d", "app")
	cmd.Dir = root
	cmd.Stdin = strings.NewReader(sql)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr

	if err := cmd.Run(); err != nil {
		return fmt.Errorf("seed rate-limit rules: %w. Is postgres running? Run 'docker compose up -d' first", err)
	}

	fmt.Println("seeded 2 rate-limit rules (wildcard '*' and /healthz)")
	return nil
}
```

```go
package ratelimit

import (
	"context"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

// RunOptions configures the rate-limit test run.
type RunOptions struct {
	Root     string
	Host     string
	Port     int
	Rate     int
	Duration string
	Paths    []string
}

// Run executes a vegeta attack against the target service.
func Run(ctx context.Context, opts RunOptions) error {
	if opts.Root == "" {
		opts.Root = "."
	}
	if opts.Host == "" {
		opts.Host = "localhost"
	}
	if opts.Port == 0 {
		opts.Port = 8080
	}
	if opts.Rate == 0 {
		opts.Rate = 200
	}
	if opts.Duration == "" {
		opts.Duration = "10s"
	}
	if len(opts.Paths) == 0 {
		opts.Paths = []string{"/healthz", "/"}
	}

	var targets []string
	for _, p := range opts.Paths {
		targets = append(targets, fmt.Sprintf("GET http://%s:%d%s", opts.Host, opts.Port, p))
	}
	targetBody := strings.Join(targets, "\n") + "\n"

	if _, err := exec.LookPath("vegeta"); err == nil {
		return runVegetaLocal(ctx, targetBody, opts)
	}
	return runVegetaDocker(ctx, opts.Root, targetBody, opts)
}

func runVegetaLocal(ctx context.Context, targets string, opts RunOptions) error {
	tmpFile := filepath.Join(os.TempDir(), "vegeta-targets.txt")
	if err := os.WriteFile(tmpFile, []byte(targets), 0o644); err != nil {
		return fmt.Errorf("write targets: %w", err)
	}
	defer os.Remove(tmpFile)

	cmd := exec.CommandContext(ctx, "vegeta", "attack",
		"-targets="+tmpFile,
		"-rate="+fmt.Sprintf("%d", opts.Rate),
		"-duration="+opts.Duration,
	)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("vegeta attack: %w", err)
	}
	return nil
}

func runVegetaDocker(ctx context.Context, root, targets string, opts RunOptions) error {
	tmpFile := filepath.Join(os.TempDir(), "vegeta-targets.txt")
	if err := os.WriteFile(tmpFile, []byte(targets), 0o644); err != nil {
		return fmt.Errorf("write targets: %w", err)
	}
	defer os.Remove(tmpFile)

	cmd := exec.CommandContext(ctx, "docker", "compose", "run", "--rm",
		"-v", tmpFile+":/targets.txt",
		"vegeta", "attack", "-targets=/targets.txt",
		"-rate="+fmt.Sprintf("%d", opts.Rate),
		"-duration="+opts.Duration,
	)
	cmd.Dir = root
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("vegeta attack via docker: %w. Is docker compose running? Run 'docker compose up -d' first", err)
	}
	return nil
}
```
